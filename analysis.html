billboard_summary <- billboard |> 
  pivot_longer(
    cols = starts_with("wk"),
    names_to = "week",
    values_to = "rank",
    values_drop_na = TRUE
  ) |> 
  mutate(week = parse_number(week)) |> 
  arrange(week) |> 
  summarize(
    first_rank = first(rank),
    best_rank = min(rank),
    first_week_at_best = week[which.min(rank)],
    total_weeks = n(),
    .by = c(artist, track)
  )

top10_songs <- billboard |> 
  pivot_longer(
    cols = starts_with("wk"),
    names_to = "week",
    values_to = "rank",
    values_drop_na = TRUE
  ) |> 
  mutate(week = parse_number(week)) |> 
  filter(min(rank) <= 10, .by = c(artist, track)) |> 
  mutate(
    notable = case_when(
      artist == "Madonna" & track == "Music" ~ "Music (Madonna)",
      artist == "Lonestar" & track == "Amazed" ~ "Amazed (Lonestar)",
      artist == "Creed" & track == "Higher" ~ "Higher (Creed)",
      .default = NA_character_
    )
  )

ggplot() +
  geom_line(
    data = filter(top10_songs, is.na(notable)),
